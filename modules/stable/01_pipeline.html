Create secret with push credentials

Note: It is strongly advised that you use ssh credentials when the option is available to you before using basic auth.

  • Select "Secrets" in the Menu

  • Click "Create" → "From YAML"

Select "Secrets" in the menu and click "Create" → "From Yaml". Copy and paste the Yaml code and hit "Save".

kind: Secret
apiVersion: v1
metadata:
  name: git-basic-auth
  namespace: user-dev
stringData:
  .git-credentials: >-
    https://user:password@gitea.apps.cluster-guid.guid.exmaple.com
  .gitconfig: >-
    [credential "https://gitea.apps.cluster-guid.guid.exmaple.com"]
      helper = store
type: Opaque

Create a pipeline

  • Click Pipelines

  • Click "Create" → "Pipeline"

Copy&paste the Yaml into the Web UI:

apiVersion: tekton.dev/v1beta1
kind: Pipeline
metadata:
  name: build-app
  namespace: user-dev
spec:
  params:
    - default: app-a
      description: 'Application name app-a, app-b, app-c'
      name: app
      type: string
  tasks:
    - name: git-clone-source
      params:
        - name: url
          value: >-
            https://gitea.apps.cluster-guid.guid.exmaple.com/user/$(params.app).git
      taskRef:
        kind: ClusterTask
        name: git-clone
      workspaces:
        - name: output
          workspace: source
    - name: build
      params:
        - name: IMAGE
          value: >-
            image-registry.openshift-image-registry.svc:5000/user-apps/$(params.app):main
        - name: DOCKERFILE
          value: ./Containerfile
      runAfter:
        - git-clone-source
      taskRef:
        kind: ClusterTask
        name: buildah
      workspaces:
        - name: source
          workspace: source

    - name: update-deployment
      params:
      - name: GIT_USER_NAME
        value: Pipeline Bot
      - name: GIT_USER_EMAIL
        value: bot@example.com
      - name: GIT_SCRIPT
        value: |
          cd $(workspaces.source.path)
          wget  https://github.com/kubernetes-sigs/kustomize/releases/download/kustomize%2Fv5.1.1/kustomize_v5.1.1_linux_amd64.tar.gz
          tar xzvf kustomize_v5.1.1_linux_amd64.tar.gz


          cd gitops/

          git config --global --add safe.directory $(pwd)

          git fetch origin
          git rebase origin/main

          # https://issues.redhat.com/browse/SRVKP-3160
          IMAGE_URL=$(echo -n "$(tasks.build.results.IMAGE_URL)" | sed 's/:main$/\@$(tasks.build.results.IMAGE_DIGEST)/' | tr -d "\n")
          $(workspaces.source.path)/kustomize  edit set image $(params.app):latest=${IMAGE_URL}

          git add kustomization.yaml
          git commit --allow-empty -m "[OpenShift Pipeline] Update $(params.app):latest deployment to $(tasks.build.results.IMAGE_DIGEST)"
          git push -v origin HEAD:main

      - name: USER_HOME
        value: /root
      - name: VERBOSE
        value: "true"
      runAfter:
      - git-clone-gitops
      taskRef:
        kind: ClusterTask
        name: git-cli
      workspaces:
      - name: source
        workspace: source
      - name: basic-auth
        workspace: git-basic-auth
    - name: git-clone-gitops
      params:
      - name: url
        value: >-
          https://gitea.apps.cluster-guid.guid.exmaple.com/user/gitops.git
      - name: revision
        value: main
      - name: refspec
        value: ""
      - name: submodules
        value: "true"
      - name: depth
        value: "0"
      - name: sslVerify
        value: "true"
      - name: crtFileName
        value: ca-bundle.crt
      - name: subdirectory
        value: gitops
      - name: sparseCheckoutDirectories
        value: ""
      - name: deleteExisting
        value: "true"
      - name: httpProxy
        value: ""
      - name: httpsProxy
        value: ""
      - name: noProxy
        value: ""
      - name: verbose
        value: "true"
      - name: userHome
        value: /home/git
      runAfter:
      - build
      taskRef:
        kind: ClusterTask
        name: git-clone
      workspaces:
      - name: output

      - name: basic-auth
        workspace: git-basic-auth

Start pipeline for Application

via CLI

Open the Web Terminal via >_

cat - > claim-template.yaml <<EOF
spec:
  accessModes:
    - ReadWriteOnce
  resources:
    requests:
      storage: 1Gi
EOF
tkn pipeline start build-app -p app=app-a -w name=source,volumeClaimTemplateFile=claim-template.yaml -w name=git-basic-auth,secret=git-basic-auth
tkn pipeline start build-app -p app=app-b -w name=source,volumeClaimTemplateFile=claim-template.yaml -w name=git-basic-auth,secret=git-basic-auth
tkn pipeline start build-app -p app=app-c -w name=source,volumeClaimTemplateFile=claim-template.yaml -w name=git-basic-auth,secret=git-basic-auth
Alternative: via WebUI

repeat this for every app-a, app-b, app-c

  • On the Pipeline detail page after creation, click "Actions" → Start

  • Fill out the form

    • app: Type onetime app-a, next time app-b and last time app-c

    • source: Select VolumeClaimTemplate

    • git-basic-auth: Select Sectet,

    • Secret: Select git-basic-auth,

  • Click "Start"

And repeat with the next app.

Create pipeline trigger

YAML

apiVersion: triggers.tekton.dev/v1beta1
kind: TriggerTemplate
metadata:
  name: build-app-pipeline-trigger-template
  namespace: user-dev
  annotations:
    triggers.tekton.dev/old-escape-quotes: 'true'
spec:
  params:
    - name: repo
      description: The Git repo name
    - name: branch
      description: The Git repo branch name
    - name: hook-type
      description: The Webhook type - branch or tag
    - name: username
      description: The user that created the branch
    - name: repo_name
      description: "Name of the repo/app: app-a, app-b or app-c"
  resourcetemplates:
    - apiVersion: tekton.dev/v1beta1
      kind: PipelineRun
      metadata:
        generateName: build-app-
      spec:
        params:
          - name: app
            value: $(tt.params.repo_name)
          - name: GIT_REPO
            value: $(tt.params.repo)
          - name: GIT_REVISION
            value: $(tt.params.branch)
          - name: PATH_CONTEXT
            value: .
          - name: VERSION
            value: openjdk-17-ubi8
        pipelineRef:
          name: build-app
        serviceAccountName: pipeline
        timeout: 1h0m0s
        workspaces:
          - name: git-basic-auth
            secret:
              secretName: git-basic-auth
          - name: source
            volumeClaimTemplate:
              spec:
                accessModes:
                  - ReadWriteOnce
                resources:
                  requests:
                    storage: 1Gi

---
apiVersion: triggers.tekton.dev/v1beta1
kind: TriggerBinding
metadata:
  name: build-app-pipeline-trigger-binding
  namespace: user-dev
spec:
  # https://docs.gitea.com/next/usage/webhooks#event-information
  params:
    - name: repo
      value: "$(body.repository.clone_url)"
    - name: repo_name
      value: "$(body.repository.name)"
    - name: branch
      value: "$(body.ref)"
    - name: hook-type
      value: "$(header.X-Gitea-Event)"
    - name: username
      value: "$(body.sender.username)"
---
apiVersion: triggers.tekton.dev/v1beta1
kind: Trigger
metadata:
  name: build-app-pipeline-trigger
  namespace: user-dev
spec:
  interceptors: []
  bindings:
  - ref: build-app-pipeline-trigger-binding
  template:
    ref: build-app-pipeline-trigger-template
---
apiVersion: triggers.tekton.dev/v1beta1
kind: EventListener
metadata:
  name: build-app-pipeline-trigger-listener
spec:
  serviceAccountName: pipeline
  triggers:
  - triggerRef: build-app-pipeline-trigger

---
apiVersion: route.openshift.io/v1
kind: Route
metadata:
  name: build-app-trigger
  namespace: user-dev
spec:
  port:
    targetPort: http-listener
  to:
    kind: Service
    name: el-build-app-pipeline-trigger-listener
    weight: 100
  wildcardPolicy: None
  tls:
    termination: edge
  insecureEdgeTerminationPolicy: Redirect
